cmake_minimum_required(VERSION 3.25)
project(MusicBoardLame LANGUAGES C)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This recipe requires the Windows x64 MSVC toolchain")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(LAME_SOURCE "" CACHE PATH "Verified unmodified LAME source directory")
if(NOT EXISTS "${LAME_SOURCE}/include/lame.h")
  message(FATAL_ERROR "LAME_SOURCE must point to the verified source tree")
endif()
file(READ "${LAME_SOURCE}/configMS.h" config)
# MSVC x64 represents long double with the same eight bytes as double.
string(REPLACE "#define SIZEOF_LONG_DOUBLE 12" "#define SIZEOF_LONG_DOUBLE 8" config "${config}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/config.h" "${config}")
set(names VbrTag bitstream encoder fft gain_analysis id3tag lame newmdct presets
  psymodel quantize quantize_pvt reservoir set_get tables takehiro util vbrquantize version mpglib_interface)
set(sources "${LAME_SOURCE}/libmp3lame/vector/xmm_quantize_sub.c")
foreach(name IN LISTS names)
  list(APPEND sources "${LAME_SOURCE}/libmp3lame/${name}.c")
endforeach()
add_library(mp3lame STATIC ${sources})
target_include_directories(mp3lame PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${LAME_SOURCE}"
  "${LAME_SOURCE}/include" "${LAME_SOURCE}/libmp3lame")
target_compile_definitions(mp3lame PRIVATE HAVE_CONFIG_H _CRT_SECURE_NO_WARNINGS)
target_compile_options(mp3lame PRIVATE /utf-8 /O2 /GS)
